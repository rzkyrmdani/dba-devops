PEMBUATAN LOG ARSIP DATABASE sqlserver

1. AKTIFKAN xp_cmdshell & CEK FOLDER

EXEC sp_configure 'show advanced options', 1; RECONFIGURE;
EXEC sp_configure 'xp_cmdshell', 1; RECONFIGURE;

2. BUAT TABEL PERMANEN UNTUK LOG AUDIT

USE master;
GO

IF OBJECT_ID('dbo.FullAuditOutput', 'U') IS NOT NULL
    DROP TABLE dbo.FullAuditOutput;
GO

CREATE TABLE dbo.FullAuditOutput (
    LogText NVARCHAR(MAX)
);

3. BUAT PROSEDUR FINAL Export_Full_Audit_ToText

IF OBJECT_ID('dbo.Export_Full_Audit_ToText', 'P') IS NOT NULL
    DROP PROCEDURE dbo.Export_Full_Audit_ToText;
GO

CREATE PROCEDURE dbo.Export_Full_Audit_ToText
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE
        @date NVARCHAR(20),
        @filePath NVARCHAR(400),
        @bcpCmd VARCHAR(8000),
        @serverName NVARCHAR(128);

    SET @date = FORMAT(GETDATE(), 'yyyyMMdd');
    SET @filePath = 'E:\BackupDB\LogDB\All_Logdb_' + @date + '.txt';
    SET @serverName = CAST(SERVERPROPERTY('ServerName') AS NVARCHAR(128));

    -- Bersihkan isi lama
    TRUNCATE TABLE dbo.FullAuditOutput;

    -- Header
    INSERT INTO dbo.FullAuditOutput (LogText)
    VALUES ('======== LOG AUDIT SQL SERVER: ' + FORMAT(GETDATE(), 'dddd, dd MMMM yyyy HH:mm:ss') + ' ========');

    ------------------------------
    -- 1. Aktifitas Default Trace
    ------------------------------
    INSERT INTO dbo.FullAuditOutput (LogText)
    SELECT
        'Event = ' + TE.name + CHAR(13) + CHAR(10) +
        'Database = ' + ISNULL(T.DatabaseName, '-') + CHAR(13) + CHAR(10) +
        'Object = ' + ISNULL(T.ObjectName, '-') + CHAR(13) + CHAR(10) +
        'User = ' + ISNULL(T.LoginName, '-') + CHAR(13) + CHAR(10) +
        'App = ' + ISNULL(T.ApplicationName, '-') + CHAR(13) + CHAR(10) +
        'Time = ' + CONVERT(NVARCHAR, T.StartTime, 120) + CHAR(13) + CHAR(10) +
        'Command = ' + CAST(ISNULL(T.TextData, '-') AS NVARCHAR(MAX)) + CHAR(13) + CHAR(10) +
        '--------------------------'
    FROM fn_trace_gettable(CONVERT(VARCHAR(255), (
        SELECT TOP 1 value FROM sys.fn_trace_getinfo(NULL) WHERE property = 2
    )), DEFAULT) T
    JOIN sys.trace_events TE ON T.EventClass = TE.trace_event_id
    WHERE T.EventClass IN (
        46, 47, 164,        -- DDL: Create, Drop, Alter
        175, 178,           -- Login, Logout
        115, 116            -- Backup database, log
    );

    ------------------------------
    -- 2. Aktifitas Session & Query
    ------------------------------
    INSERT INTO dbo.FullAuditOutput (LogText)
    SELECT
        'session_id = ' + CAST(s.session_id AS NVARCHAR) + CHAR(13) + CHAR(10) +
        'username = ' + ISNULL(s.login_name, '-') + CHAR(13) + CHAR(10) +
        'host_name = ' + ISNULL(s.host_name, '-') + CHAR(13) + CHAR(10) +
        'program_name = ' + ISNULL(s.program_name, '-') + CHAR(13) + CHAR(10) +
        'database_name = ' + ISNULL(DB_NAME(r.database_id), '-') + CHAR(13) + CHAR(10) +
        'status = ' + ISNULL(r.status, '-') + CHAR(13) + CHAR(10) +
        'command = ' + ISNULL(r.command, '-') + CHAR(13) + CHAR(10) +
        'query = ' + ISNULL(REPLACE(t.text, CHAR(13) + CHAR(10), ' '), '-') + CHAR(13) + CHAR(10) +
        'cpu_time = ' + CAST(ISNULL(r.cpu_time, 0) AS NVARCHAR) + ' ms' + CHAR(13) + CHAR(10) +
        'duration = ' + CAST(ISNULL(r.total_elapsed_time, 0) AS NVARCHAR) + ' ms' + CHAR(13) + CHAR(10) +
        'wait_type = ' + ISNULL(r.wait_type, '-') + CHAR(13) + CHAR(10) +
        'wait_time = ' + CAST(ISNULL(r.wait_time, 0) AS NVARCHAR) + ' ms' + CHAR(13) + CHAR(10) +
        'client_ip = ' + ISNULL(c.client_net_address, '-') + CHAR(13) + CHAR(10) +
        'last_start = ' + ISNULL(CONVERT(NVARCHAR, s.last_request_start_time, 120), '-') + CHAR(13) + CHAR(10) +
        'last_end = ' + ISNULL(CONVERT(NVARCHAR, s.last_request_end_time, 120), '-') + CHAR(13) + CHAR(10) +
        'blocking_session_id = ' + CAST(ISNULL(r.blocking_session_id, 0) AS NVARCHAR) + CHAR(13) + CHAR(10) +
        '----------------------------'
    FROM sys.dm_exec_sessions s
    LEFT JOIN sys.dm_exec_requests r ON s.session_id = r.session_id
    LEFT JOIN sys.dm_exec_connections c ON s.session_id = c.session_id
    OUTER APPLY sys.dm_exec_sql_text(r.sql_handle) t
    WHERE s.is_user_process = 1;

    ------------------------------
    -- 3. Export ke File .txt
    ------------------------------
    SET @bcpCmd = 'bcp "SELECT LogText FROM master.dbo.FullAuditOutput" queryout "' + @filePath + '" -c -T -S "' + @serverName + '"';
    EXEC xp_cmdshell @bcpCmd;
END;
GO

USE master;

IF OBJECT_ID('dbo.Export_Full_Audit_ToText', 'P') IS NOT NULL
    DROP PROCEDURE dbo.Export_Full_Audit_ToText;
GO

CREATE PROCEDURE dbo.Export_Full_Audit_ToText
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE 
        @date NVARCHAR(20),
        @filePath NVARCHAR(400),
        @bcpCmd VARCHAR(8000),
        @serverName NVARCHAR(128);

    SET @date = FORMAT(GETDATE(), 'yyyyMMdd');
    SET @filePath = 'C:\BACKUP\arsip_logdb\All_Logdb_' + @date + '.txt';
    SET @serverName = CAST(SERVERPROPERTY('ServerName') AS NVARCHAR(128));

    -- Pastikan tabel log ada
    IF OBJECT_ID('dbo.FullAuditOutput', 'U') IS NULL
    BEGIN
        CREATE TABLE dbo.FullAuditOutput (
            LogText NVARCHAR(MAX)
        );
    END

    -- Kosongkan isi tabel
    TRUNCATE TABLE dbo.FullAuditOutput;

    -- Header
    INSERT INTO dbo.FullAuditOutput (LogText)
    VALUES ('======== ALL LOG DB SQL SERVER: ' + FORMAT(GETDATE(), 'dddd, dd MMMM yyyy HH:mm:ss') + ' ========' + CHAR(13) + CHAR(10));

    --------------------------------------------
    -- Bagian 1: Event DDL, Login, Backup (Trace)
    --------------------------------------------
    INSERT INTO dbo.FullAuditOutput (LogText)
    SELECT 
        'Event = ' + TE.name + CHAR(13) + CHAR(10) +
        'Database = ' + ISNULL(T.DatabaseName, '-') + CHAR(13) + CHAR(10) +
        'Object = ' + ISNULL(T.ObjectName, '-') + CHAR(13) + CHAR(10) +
        'User = ' + ISNULL(T.LoginName, '-') + CHAR(13) + CHAR(10) +
        'App = ' + ISNULL(T.ApplicationName, '-') + CHAR(13) + CHAR(10) +
        'Time = ' + CONVERT(NVARCHAR, T.StartTime, 120) + CHAR(13) + CHAR(10) +
        'Command = ' + ISNULL(CAST(T.TextData AS NVARCHAR(MAX)), '-') + CHAR(13) + CHAR(10) +
        '--------------------------' + CHAR(13) + CHAR(10)
    FROM fn_trace_gettable(
            CONVERT(VARCHAR(255), (
                SELECT TOP 1 value 
                FROM sys.fn_trace_getinfo(NULL) 
                WHERE property = 2
            )), DEFAULT) T
    JOIN sys.trace_events TE ON T.EventClass = TE.trace_event_id
    WHERE T.EventClass IN (46, 47, 164, 175, 178, 115, 116)
      AND CONVERT(DATE, T.StartTime) = CONVERT(DATE, GETDATE());

    --------------------------------------------
    -- Bagian 2: Session Aktif Hari Ini
    --------------------------------------------
    INSERT INTO dbo.FullAuditOutput (LogText)
    SELECT 
        'session_id = ' + CAST(s.session_id AS NVARCHAR) + CHAR(13) + CHAR(10) +
        'username = ' + ISNULL(s.login_name, '-') + CHAR(13) + CHAR(10) +
        'host_name = ' + ISNULL(s.host_name, '-') + CHAR(13) + CHAR(10) +
        'program_name = ' + ISNULL(s.program_name, '-') + CHAR(13) + CHAR(10) +
        'database_name = ' + ISNULL(DB_NAME(r.database_id), '-') + CHAR(13) + CHAR(10) +
        'status = ' + ISNULL(r.status, '-') + CHAR(13) + CHAR(10) +
        'command = ' + ISNULL(r.command, '-') + CHAR(13) + CHAR(10) +
        'query = ' + ISNULL(REPLACE(CAST(t.text AS NVARCHAR(MAX)), CHAR(13) + CHAR(10), ' '), '-') + CHAR(13) + CHAR(10) +
        'cpu_time = ' + CAST(ISNULL(r.cpu_time, 0) AS NVARCHAR) + ' ms' + CHAR(13) + CHAR(10) +
        'duration = ' + CAST(ISNULL(r.total_elapsed_time, 0) AS NVARCHAR) + ' ms' + CHAR(13) + CHAR(10) +
        'wait_type = ' + ISNULL(r.wait_type, '-') + CHAR(13) + CHAR(10) +
        'wait_time = ' + CAST(ISNULL(r.wait_time, 0) AS NVARCHAR) + ' ms' + CHAR(13) + CHAR(10) +
        'client_ip = ' + ISNULL(c.client_net_address, '-') + CHAR(13) + CHAR(10) +
        'last_start = ' + ISNULL(CONVERT(NVARCHAR, s.last_request_start_time, 120), '-') + CHAR(13) + CHAR(10) +
        'last_end = ' + ISNULL(CONVERT(NVARCHAR, s.last_request_end_time, 120), '-') + CHAR(13) + CHAR(10) +
        'blocking_session_id = ' + CAST(ISNULL(r.blocking_session_id, 0) AS NVARCHAR) + CHAR(13) + CHAR(10) +
        '----------------------------' + CHAR(13) + CHAR(10)
    FROM sys.dm_exec_sessions s
    LEFT JOIN sys.dm_exec_requests r ON s.session_id = r.session_id
    LEFT JOIN sys.dm_exec_connections c ON s.session_id = c.session_id
    OUTER APPLY sys.dm_exec_sql_text(r.sql_handle) t
    WHERE s.is_user_process = 1
      AND CONVERT(DATE, s.last_request_start_time) = CONVERT(DATE, GETDATE());

    --------------------------------------------
    -- Export ke File TXT
    --------------------------------------------
    SET @bcpCmd = 'bcp "SELECT LogText FROM master.dbo.FullAuditOutput" queryout "' + @filePath + '" -c -T -S "' + @serverName + '"';
    EXEC xp_cmdshell @bcpCmd;
END;
GO


4. JALANKAN PROSEDUR

EXEC dbo.Export_Full_Audit_ToText;
