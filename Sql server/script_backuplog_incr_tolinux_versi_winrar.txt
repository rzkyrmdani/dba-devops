:: === BERIKUT SCRIPT BACKUP INCREMENTAL LOGALL DB YANG TERDAFTAR ZIP SCP KE LINUX VERSI WINRAR ===

@echo off
setlocal

:: === KONFIGURASI ===
set "SQL_USER=backup_db"
set "SQL_PASS=Plnpusat@2024!"
set "SQL_SERVER=10.1.8.205\RDBSILM"
set "BACKUP_DIR=E:\DATABASE\BACKUP\ALL_BACKUP_LOG_ZIP"
set "SQLCMD_PATH=C:\Program Files\Microsoft SQL Server\110\Tools\Binn\SQLCMD.EXE"
set "PSCP_PATH=C:\Program Files\PuTTY\pscp.exe"
set "WINRAR_PATH=C:\Program Files\WinRAR\WinRAR.exe"
set "SCP_PORT=6969"
set "SCP_USER=icon"
set "SCP_PASS=R4in!Sc0mFOrT1nG"
set "SCP_IP=10.14.152.62"
set "SCP_TARGET=/Backup_Database/Database/production/silm_backuplog"

:: === BUAT FOLDER BACKUP JIKA BELUM ADA ===
if not exist "%BACKUP_DIR%" (
    mkdir "%BACKUP_DIR%"
)

:: === TANGGAL DAN WAKTU UNTUK NAMA FILE ===
for /f %%i in ('powershell -NoLogo -Command "Get-Date -Format yyyyMMdd_HHmmss"') do set "DATETIME=%%i"
set "ZIP_NAME=BackupLog_ALL_DBSILM_%DATETIME%.zip"
set "ZIP_PATH=%BACKUP_DIR%\%ZIP_NAME%"

echo [INFO] Memulai backup TRANSACTION LOG untuk semua database...

:: === BACKUP TRANSACTION LOG SEMUA DATABASE ===
"%SQLCMD_PATH%" -S %SQL_SERVER% -U %SQL_USER% -P %SQL_PASS% -Q ^
"DECLARE @name NVARCHAR(256), @fileName NVARCHAR(512), @fileDate NVARCHAR(20), @path NVARCHAR(512); ^
SET @path = N'%BACKUP_DIR%\'; ^
SET @fileDate = CONVERT(VARCHAR(20), GETDATE(), 112) + '_' + REPLACE(CONVERT(VARCHAR(8), GETDATE(), 108), ':', ''); ^
DECLARE db_cursor CURSOR FOR ^
    SELECT name FROM sys.databases ^
    WHERE name NOT IN ('master','model','msdb','tempdb') AND recovery_model_desc IN ('FULL','BULK_LOGGED'); ^
OPEN db_cursor FETCH NEXT FROM db_cursor INTO @name ^
WHILE @@FETCH_STATUS = 0 ^
BEGIN SET @fileName = @path + @name + '_TLOG_' + @fileDate + '.trn'; ^
EXEC('BACKUP LOG [' + @name + '] TO DISK = ''' + @fileName + ''' WITH INIT, COMPRESSION'); ^
FETCH NEXT FROM db_cursor INTO @name END ^
CLOSE db_cursor DEALLOCATE db_cursor"

:: === CEK FILE TRN ADA ===
if not exist "%BACKUP_DIR%\*.trn" (
    echo [WARNING] Tidak ada file TRN ditemukan. Proses dibatalkan.
    exit /b
)

:: === ZIP FILE .TRN DENGAN WINRAR ===
echo [INFO] Mengompres file TRN dengan WinRAR...
"%WINRAR_PATH%" a -afzip -ep1 -inul "%ZIP_PATH%" "%BACKUP_DIR%\*.trn"

:: === HAPUS SEMUA FILE .TRN SETELAH ZIP ===
del /q "%BACKUP_DIR%\*.trn"

:: === KIRIM FILE ZIP KE SERVER UBUNTU ===
echo [INFO] Mengirim file: %ZIP_NAME%
echo y | "%PSCP_PATH%" -P %SCP_PORT% -pw "%SCP_PASS%" "%ZIP_PATH%" %SCP_USER%@%SCP_IP%:%SCP_TARGET%

:: === SELESAI ===
echo [SELESAI] Backup Transaction Log dan transfer berhasil.
timeout /t 3 /nobreak >nul
exit
