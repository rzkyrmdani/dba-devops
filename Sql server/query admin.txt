CREATE OR ALTER PROCEDURE InsertTenUsers
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @i INT = 1;

    WHILE @i <= 10
    BEGIN
        INSERT INTO [user] (username, password, email, full_name, updated_at)
        VALUES (
            'auto_user_' + CAST(@i AS NVARCHAR(10)) + '_' + FORMAT(GETDATE(), 'yyyyMMdd_HHmmss'),
            'hashed_password_' + CAST(@i AS NVARCHAR(10)),
            'auto_user_' + CAST(@i AS NVARCHAR(10)) + '_' + FORMAT(GETDATE(), 'yyyyMMdd_HHmmss') + '@example.com',
            'Auto User ' + CAST(@i AS NVARCHAR(10)),
            GETDATE()
        );
        SET @i = @i + 1;
    END
END;


EXEC InsertTenUsers;




SELECT 
    t.name AS TableName,
    SUM(ps.row_count) AS [RowCount]
FROM 
    sys.tables t
JOIN 
    sys.dm_db_partition_stats ps 
    ON t.object_id = ps.object_id
WHERE 
    ps.index_id IN (0, 1) -- Hanya menghitung row dari heap atau clustered index
GROUP BY 
    t.name
ORDER BY 
    t.name;

USE msdb;
GO
SELECT 
    bs.database_name AS DatabaseName,
    bs.backup_start_date AS BackupStartDate,
    bs.backup_finish_date AS BackupFinishDate,
    CASE bs.type
        WHEN 'D' THEN 'Full'
        WHEN 'I' THEN 'Differential'
        WHEN 'L' THEN 'Transaction Log'
        ELSE 'Unknown'
    END AS BackupType,
    bs.backup_size / 1024 / 1024 AS BackupSizeMB,
    bmf.physical_device_name AS BackupFile
FROM 
    backupset bs
JOIN 
    backupmediafamily bmf 
    ON bs.media_set_id = bmf.media_set_id
WHERE 
    bs.database_name = 'ProRBA2PLN' -- Filter hanya database yg dituju
ORDER BY 
    bs.backup_finish_date DESC;
