CREATE OR ALTER PROCEDURE InsertTenUsers
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @i INT = 1;

    WHILE @i <= 10
    BEGIN
        INSERT INTO [user] (username, password, email, full_name, updated_at)
        VALUES (
            'auto_user_' + CAST(@i AS NVARCHAR(10)) + '_' + FORMAT(GETDATE(), 'yyyyMMdd_HHmmss'),
            'hashed_password_' + CAST(@i AS NVARCHAR(10)),
            'auto_user_' + CAST(@i AS NVARCHAR(10)) + '_' + FORMAT(GETDATE(), 'yyyyMMdd_HHmmss') + '@example.com',
            'Auto User ' + CAST(@i AS NVARCHAR(10)),
            GETDATE()
        );
        SET @i = @i + 1;
    END
END;

SELECT 
    t.name AS TableName,
    SUM(ps.row_count) AS [RowCount]
FROM 
    sys.tables t
JOIN 
    sys.dm_db_partition_stats ps 
    ON t.object_id = ps.object_id
WHERE 
    ps.index_id IN (0, 1) -- Hanya menghitung row dari heap atau clustered index
GROUP BY 
    t.name
ORDER BY 
    t.name;

SELECT 
    bs.database_name,
    bs.backup_start_date,
    bs.backup_finish_date,
    bs.type AS backup_type,
    CASE bs.type
        WHEN 'D' THEN 'Full'
        WHEN 'I' THEN 'Differential'
        WHEN 'L' THEN 'Transaction Log'
        ELSE 'Other'
    END AS backup_type_desc,
    CONVERT(DECIMAL(18,2), bs.backup_size / 1048576.0) AS backup_size_MB,
    bmf.physical_device_name
FROM 
    msdb.dbo.backupset bs
INNER JOIN 
    msdb.dbo.backupmediafamily bmf ON bs.media_set_id = bmf.media_set_id
ORDER BY 
    bs.backup_finish_date DESC;

SELECT 
    bs.database_name AS DatabaseName,
    bs.backup_start_date AS BackupStartDate,
    bs.backup_finish_date AS BackupFinishDate,
    CASE bs.type
        WHEN 'D' THEN 'Full'
        WHEN 'I' THEN 'Differential'
        WHEN 'L' THEN 'Transaction Log'
        ELSE 'Unknown'
    END AS BackupType,
    bs.backup_size / 1024 / 1024 AS BackupSizeMB,
    bmf.physical_device_name AS BackupFile
FROM 
    backupset bs
JOIN 
    backupmediafamily bmf 
    ON bs.media_set_id = bmf.media_set_id
WHERE 
    bs.database_name = 'ProRBA2PLN' -- Filter hanya database yg dituju
ORDER BY 
    bs.backup_finish_date DESC;

---------------- cek total ukuran database yang terdaftar ----------------


SELECT 
    d.name AS DatabaseName,
    CAST(SUM(f.size * 8 / 1024.0) AS DECIMAL(10,2)) AS TotalSizeMB,
    CAST(SUM(f.size * 8 / 1024.0 / 1024) AS DECIMAL(10,2)) AS TotalSizeGB
FROM sys.databases d
JOIN sys.master_files f ON d.database_id = f.database_id
GROUP BY d.name
ORDER BY TotalSizeMB DESC;

-----------------------------cek tablespace
SELECT     
    db.name AS database_name,
    fg.name AS filegroup_name,
    mf.name AS logical_file_name,
    mf.physical_name,
    mf.type_desc AS file_type,
    mf.size * 8 / 1024 AS size_MB,
    mf.max_size,
    mf.growth,
    mf.is_percent_growth
FROM     
    sys.master_files mf
JOIN     
    sys.databases db ON mf.database_id = db.database_id
LEFT JOIN     
    sys.filegroups fg ON mf.data_space_id = fg.data_space_id
WHERE
    db.name = 'ProRBA2PLN'
ORDER BY     
    db.name, fg.name;

-- Backup log
SELECT
    backup_finish_date AS event_time,
    CASE bs.type
        WHEN 'D' THEN 'BACKUP DATABASE'
        WHEN 'L' THEN 'BACKUP LOG'
        WHEN 'I' THEN 'BACKUP DIFFERENTIAL'
        ELSE 'BACKUP OTHER'
    END AS action_type,
    1 AS succeeded,
    bs.user_name AS username,
    bs.database_name,
    NULL AS object_name,
    bmf.physical_device_name AS statement_or_device,
    'BACKUP' AS source
FROM msdb.dbo.backupset bs
JOIN msdb.dbo.backupmediafamily bmf
    ON bs.media_set_id = bmf.media_set_id
WHERE bs.backup_finish_date >= DATEADD(DAY, -1, GETDATE()) -- log 1 hari terakhir
