- CARA KILL/DISCONNECT PADA USER YG SEDANG LOGIN
1. buat 2 user dulu 
	--> create user JONI identified by joni123;
	--> create user JONO identified by jono123;
2. kasih hak ases login
	--> grant create session to JONI;
	--> grant create session to JONO;
3. kasih akses sistem pada user JONI
	--> grant alter system to JONI;
4. Cek ID user dari user SYSADMIN
	--> select sid, serial#, status, username from v$session WHERE username IS NOT NULL ORDER by username ASC;
5. setelah dapat ID lakukan di user JONI
	--> loign user JONI
	--> alter system disconnect session '<SID>, <SERIAL>' immediate;
	--> atau ALTER SYSTEM KILL SESSION '<SID>, <SERIAL>' 
	


CARA USER 1 KILL/DISCONNECT USER 2 DAN USER 2 UPDATE ROLE DAN PROFILE
1. buat 2 user
	--> create user QUSAY identified by qusay123; (user 1)
	--> create user BAGAS identified by bagas123; (user 2)
2. kasih hak akses login
	--> grant create session to BAGAS, QUSAY;
3. buat role di sys
	--> create role COBA_ROLE;
4. kasih role ke user 2 (BAGAS) di user SYS
	--> grant COBA_ROLE to BAGAS;
5. kasih akses buat table pada role di user SYS
	--> grant create table to COBA_ROLE with admin option;
6. Cek/Melihat akses apa yg diberikan utk ROLE
	--> select * from role_sys_privs where role='COBA_ROLE' order by 1;
7. login user 2 (BAGAS) dan coba buat table
	--> create table KARYAWAN (id varchar(10), nama varchar(20), kota varchar(25), email varchar(25));
	--> insert into KARYAWAN(id, nama, kota, email) values('1', 'beni', 'solo', 'beni@mail.com');
	--> select * from KARYAWAN
8. di user 2 (bagas) cabut hak akses buat table pada role
	--> revoke create table from COBA_ROLE;
	--> coba buat table kembali
9. sekarang utk update profile pada user 2(bagas). Buat profile di sys
	--> create profile BENI_PROFILE limit failed_login_attempts 2;
10. barikan akses update ke user 2 (bagas) dari sys
	--> grant alter profile to BAGAS;
11. di user bagas lakukan update profile
	--> alter profile BENI_PROFILE limit failed_login_attempts 1;
