- CEK STATUS DATABASE

SELECT NAME, OPEN_MODE, DATABASE_ROLE, PROTECTION_MODE, PROTECTION_LEVEL FROM V$DATABASE;

output:

NAME → Nama database
OPEN_MODE → Status DB (READ WRITE / MOUNTED)
DATABASE_ROLE → PRIMARY / STANDBY
PROTECTION_MODE → Mode proteksi

- Status Instance

SELECT INSTANCE_NAME, STATUS, DATABASE_STATUS FROM V$INSTANCE;

- Status Proses Data Guard (ODG oracle data guard)

SELECT PROCESS, STATUS, THREAD#, SEQUENCE# FROM V$MANAGED_STANDBY;

Normal:

PRIMARY → ada LNS
STANDBY → ada MRP0 APPLYING_LOG

- Cek Lag (delay odg)

SELECT NAME, VALUE, UNIT FROM V$DATAGUARD_STATS;

Parameter penting:

transport lag
apply lag

👉 Ideal: 0 atau kecil

- Cek Archive Log Gap

SELECT * FROM V$ARCHIVE_GAP;

👉 Jika ada hasil → ❌ log missing (Harus kosong karena kosong berarti normal)

- Cek Status Destination Log

SELECT DEST_ID, STATUS, ERROR FROM V$ARCHIVE_DEST;

👉 Harus:

STATUS = VALID
ERROR = NULL

- Detail Status Data Guard 

SELECT DEST_ID, STATUS, TARGET, ERROR FROM V$ARCHIVE_DEST_STATUS;

- CEK List Archive Log

SELECT SEQUENCE#, FIRST_TIME, NEXT_TIME, APPLIED FROM V$ARCHIVED_LOG ORDER BY SEQUENCE# DESC;

- Bandingkan Primary vs Standby

SELECT THREAD#, MAX(SEQUENCE#) FROM V$ARCHIVED_LOG GROUP BY THREAD#;

- Cek performance TOP query (berdasarkan cpu)

SELECT SQL_ID, EXECUTIONS, CPU_TIME FROM V$SQLAREA ORDER BY CPU_TIME DESC FETCH FIRST 10 ROWS ONLY;

- Cek performance TOP query (berdasarkan wait event)

SELECT EVENT, TOTAL_WAITS, TIME_WAITED FROM V$SYSTEM_EVENT ORDER BY TIME_WAITED DESC;

- Query BERJALAN

SELECT SID, SQL_ID, STATUS, EVENT FROM V$SESSION WHERE STATUS='ACTIVE';

- QUery berjalan dan blocking

SELECT BLOCKING_SESSION, SID, SERIAL# FROM V$SESSION WHERE BLOCKING_SESSION IS NOT NULL;

- cek archive log

SELECT SEQUENCE#, FIRST_TIME, NEXT_TIME, APPLIED FROM V$ARCHIVED_LOG ORDER BY SEQUENCE# DESC;

- hapus archive log yg lebih dari 2 hari

DELETE NOPROMPT ARCHIVELOG UNTIL TIME "SYSDATE-2";

DELETE NOPROMPT ARCHIVELOG ALL COMPLETED BEFORE 'SYSDATE-2' APPLIED ON ALL STANDBY; (Khusus ODG)
