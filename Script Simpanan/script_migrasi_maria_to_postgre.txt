
SKRIP CONVERT MARIADB TO POSTGRESQL

========================== untuk table ambil dari mariadb convert ke postgresql =========================================

SELECT
    CONCAT(
        '-- =======================================================\n',
        '-- Skrip CREATE TABLE untuk PostgreSQL\n',
        '-- Tabel: ',
        table_name,
        '\n-- =======================================================\n',
        'CREATE TABLE public."', table_name, '" (\n',
        GROUP_CONCAT(column_definition ORDER BY ordinal_position SEPARATOR ',\n'),
        CASE
            WHEN MAX(pk_columns) IS NOT NULL THEN CONCAT(',\n    PRIMARY KEY (', MAX(pk_columns), ')')
            ELSE ''
        END,
        '\n);\n\n'
    ) AS postgresql_create_script
FROM
    (
        SELECT
            c.table_name,
            c.ordinal_position,
            CONCAT(
                '    "', c.column_name, '" ',
                -- Konversi Tipe Data
                CASE
                    WHEN c.extra = 'auto_increment' AND c.data_type LIKE '%int' THEN 
                        CASE c.data_type WHEN 'bigint' THEN 'BIGSERIAL' ELSE 'SERIAL' END
                    WHEN c.data_type = 'int' THEN 'INTEGER'
                    WHEN c.data_type = 'bigint' THEN 'BIGINT'
                    WHEN c.data_type = 'tinyint' AND c.column_type LIKE '%(1)%' THEN 'BOOLEAN'
                    WHEN c.data_type = 'tinyint' THEN 'SMALLINT'
                    WHEN c.data_type IN ('varchar', 'char') THEN CONCAT(UPPER(c.data_type), '(', c.character_maximum_length, ')')
                    WHEN c.data_type IN ('text', 'longtext', 'mediumtext') THEN 'TEXT'
                    WHEN c.data_type IN ('datetime', 'timestamp') THEN 'TIMESTAMP'
                    WHEN c.data_type = 'date' THEN 'DATE'
                    WHEN c.data_type = 'decimal' THEN CONCAT('DECIMAL(', c.numeric_precision, ', ', c.numeric_scale, ')')
                    WHEN c.data_type = 'double' THEN 'DOUBLE PRECISION'
                    WHEN c.data_type = 'float' THEN 'REAL'
                    WHEN c.data_type = 'json' THEN 'JSONB'
                    ELSE UPPER(c.data_type)
                END,
                -- Nullability
                IF(c.is_nullable = 'NO' AND c.extra <> 'auto_increment', ' NOT NULL', ''),
                -- Default Value
                IF(c.column_default IS NOT NULL AND c.extra <> 'auto_increment', CONCAT(' DEFAULT ', 
                    CASE
                        WHEN c.column_default = 'CURRENT_TIMESTAMP' THEN 'CURRENT_TIMESTAMP'
                        WHEN c.data_type IN ('int', 'bigint', 'decimal', 'float', 'double') THEN c.column_default
                        ELSE QUOTE(c.column_default)
                    END
                ), '')
            ) AS column_definition,
            IF(c.column_key = 'PRI', CONCAT('"', c.column_name, '"'), NULL) AS pk_columns
        FROM
            information_schema.columns c
        WHERE
            c.table_schema = @db_name
    ) AS column_details
GROUP BY
    table_name
ORDER BY
    table_name;

------------ untuk insert ambil dari per table pada mariadb convert ke postgresql



SELECT
    CONCAT(
        -- Perintah ini akan membuat skrip INSERT final
        "SELECT CONCAT('INSERT INTO public.\"",
        table_name,
        "\" (",
        -- Menggabungkan semua nama kolom
        GROUP_CONCAT(DISTINCT CONCAT('\"', column_name, '\"') ORDER BY ordinal_position SEPARATOR ', '),
        ") VALUES (', ",
        -- Menggabungkan semua nilai kolom, meng-handle NULL dan karakter kutip (')
        GROUP_CONCAT(CONCAT("COALESCE(QUOTE(`", column_name, "`), 'NULL')") ORDER BY ordinal_position SEPARATOR ", ', ', "),
        ", ');') AS insert_statement FROM `",
        table_schema, "`.`", table_name, "`;"
    ) AS command_to_generate_inserts
FROM
    information_schema.columns
WHERE
    -- GANTI NAMA DATABASE DI SINI
    table_schema = 'disjaya_dashboard'
GROUP BY
    table_schema, table_name
ORDER BY
    table_name;

-------------- compare data mariadb dan postgresql

mariadb

SELECT 
    table_schema AS database_name,
    COUNT(table_name) AS total_tables,
    SUM(table_rows) AS total_rows,
    ROUND(SUM(data_length + index_length) / 1024 / 1024, 2) AS total_size_mb
FROM information_schema.tables
WHERE table_schema = 'disjaya_dashboard'
GROUP BY table_schema;


postgresql

SELECT
    current_database() AS database_name,
    COUNT(c.oid) AS total_tables,
    SUM(c.reltuples)::BIGINT AS total_rows,  -- estimasi total row
    ROUND(SUM(pg_total_relation_size(c.oid)) / 1024 / 1024, 2) AS total_size_mb
FROM pg_class c
JOIN pg_namespace n ON n.oid = c.relnamespace
WHERE c.relkind = 'r'  -- hanya tabel
  AND n.nspname NOT IN ('pg_catalog', 'information_schema')
GROUP BY current_database();
