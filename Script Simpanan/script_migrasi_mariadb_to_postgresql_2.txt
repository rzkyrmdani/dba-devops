============================== Tahap 1: Skrip Migrasi Skema (table mariadb ke postgresql ==================================================

SELECT
    CONCAT(
        '-- Skrip CREATE TABLE untuk Tabel: ',
        table_name,
        '\nCREATE TABLE public."', table_name, '" (\n',
        GROUP_CONCAT(column_definition ORDER BY ordinal_position SEPARATOR ',\n'),
        -- Menambahkan Primary Key constraint di akhir jika belum didefinisikan di kolom SERIAL
        CASE
            WHEN MAX(pk_columns) IS NOT NULL AND MAX(has_serial_pk) = 0 THEN CONCAT(',\n    PRIMARY KEY (', MAX(pk_columns), ')')
            ELSE ''
        END,
        '\n);\n'
    ) AS postgresql_create_script
FROM
    (
        SELECT
            c.table_name,
            c.ordinal_position,
            CONCAT(
                '    "', c.column_name, '" ',
                -- Konversi Tipe Data
                CASE
                    WHEN c.extra = 'auto_increment' AND c.data_type LIKE '%int' THEN 
                        CASE c.data_type WHEN 'bigint' THEN 'BIGSERIAL PRIMARY KEY' ELSE 'SERIAL PRIMARY KEY' END
                    WHEN c.data_type = 'int' THEN 'INTEGER'
                    WHEN c.data_type = 'bigint' THEN 'BIGINT'
                    WHEN c.data_type = 'tinyint' AND c.column_type LIKE '%(1)%' THEN 'BOOLEAN'
                    WHEN c.data_type = 'tinyint' THEN 'SMALLINT'
                    WHEN c.data_type IN ('varchar', 'char') THEN CONCAT(UPPER(c.data_type), '(', c.character_maximum_length, ')')
                    WHEN c.data_type IN ('text', 'longtext', 'mediumtext') THEN 'TEXT'
                    WHEN c.data_type IN ('datetime', 'timestamp') THEN 'TIMESTAMP'
                    WHEN c.data_type = 'date' THEN 'DATE'
                    WHEN c.data_type = 'decimal' THEN CONCAT('DECIMAL(', c.numeric_precision, ', ', c.numeric_scale, ')')
                    WHEN c.data_type = 'double' THEN 'DOUBLE PRECISION'
                    WHEN c.data_type = 'float' THEN 'REAL'
                    WHEN c.data_type = 'json' THEN 'JSONB'
                    ELSE UPPER(c.data_type)
                END,
                -- Nullability
                IF(c.is_nullable = 'NO' AND c.extra <> 'auto_increment', ' NOT NULL', ''),
                -- Default Value
                IF(c.column_default IS NOT NULL AND c.extra <> 'auto_increment', CONCAT(' DEFAULT ', 
                    CASE
                        WHEN c.column_default = 'CURRENT_TIMESTAMP' THEN 'CURRENT_TIMESTAMP'
                        WHEN c.data_type IN ('int', 'bigint', 'decimal', 'float', 'double') THEN c.column_default
                        ELSE QUOTE(c.column_default)
                    END
                ), '')
            ) AS column_definition,
            IF(c.column_key = 'PRI', CONCAT('"', c.column_name, '"'), NULL) AS pk_columns,
            IF(c.extra = 'auto_increment' AND c.column_key = 'PRI', 1, 0) as has_serial_pk
        FROM
            information_schema.columns c
        WHERE
            c.table_schema = 'disjaya_dashboard'
    ) AS column_details
GROUP BY
    table_name
ORDER BY
    table_name;
	
	
	
============================================== Tahap 2: Skrip Migrasi Data ==================================================

#!/bin/bash

# ===============================================================
# PENGATURAN KONEKSI DATABASE
# ===============================================================
DB_USER="root"
DB_PASS="Welcome1"
DB_NAME="disjaya_dashboard"
PG_USER="postgres"
PG_DB_NAME="disjaya_dashboard"
# ===============================================================
# PENGATURAN FILE & DIREKTORI
# ===============================================================
EXPORT_DIR="/tmp/db_migration_data"
FINAL_PG_SCRIPT="final_restore_script.sql"
# ===============================================================

# Membuat direktori untuk menyimpan data sementara
mkdir -p "${EXPORT_DIR}"
echo "Direktori ekspor dibuat di ${EXPORT_DIR}"

# Mengosongkan file skrip final jika sudah ada
> "${FINAL_PG_SCRIPT}"

echo "Memulai proses ekspor dari MariaDB dan membuat skrip restore PostgreSQL..."

# Mendapatkan daftar semua tabel dari database
TABLES=$(mysql -u"${DB_USER}" -p"${DB_PASS}" -D"${DB_NAME}" -N -s -e "SHOW TABLES;")

if [ $? -ne 0 ]; then
    echo "Gagal mendapatkan daftar tabel. Periksa koneksi MariaDB Anda."
    exit 1
fi

# Loop untuk setiap tabel
for table in ${TABLES}; do
    echo "  - Memproses tabel: ${table}"
    
    # Menentukan path file TSV (Tab-Separated, lebih aman dari CSV)
    TSV_FILE_PATH="${EXPORT_DIR}/${table}.tsv"
    
    # 1. Ekspor data dari MariaDB ke file TSV
    # Opsi -B (batch mode) akan menggunakan tab sebagai pemisah
    mysql -u"${DB_USER}" -p"${DB_PASS}" -D"${DB_NAME}" -B -e "SELECT * FROM \`${table}\`;" > "${TSV_FILE_PATH}"
    
    # 2. Buat perintah \copy untuk PostgreSQL dan tambahkan ke skrip final
    # Perintah \copy adalah cara psql untuk bulk import yang sangat cepat
    echo "-- Memuat data untuk tabel: ${table}" >> "${FINAL_PG_SCRIPT}"
    echo "\\copy public.\"${table}\" FROM '${TSV_FILE_PATH}' WITH (FORMAT text, NULL 'NULL');" >> "${FINAL_PG_SCRIPT}"
    echo "" >> "${FINAL_PG_SCRIPT}"
done

echo ""
echo "✅ SELESAI!"
echo "Semua data telah diekspor ke direktori: ${EXPORT_DIR}"
echo "Skrip restore untuk PostgreSQL telah dibuat di file: ${FINAL_PG_SCRIPT}"
echo ""
echo "Untuk menjalankan restore, pastikan file TSV dapat diakses dari server PostgreSQL, lalu jalankan:"
echo "psql -U ${PG_USER} -d ${PG_DB_NAME} -f ${FINAL_PG_SCRIPT}"




================================ Tahap 3: Migrasi Objek Final dan Logika Bisnis ==================================================================

Views, Functions, Procedures, ,Triggers dan Index

-------- Cek Views
SELECT table_name, view_definition 
FROM information_schema.views 
WHERE table_schema = 'disjaya_dashboard';

-------- Cek Functions & Procedures
SELECT ROUTINE_NAME, ROUTINE_TYPE, ROUTINE_DEFINITION 
FROM information_schema.routines 
WHERE routine_schema = 'disjaya_dashboard';

-------- Cek Triggers
SELECT trigger_name, event_manipulation, action_statement 
FROM information_schema.triggers 
WHERE trigger_schema = 'disjaya_dashboard';

-------- cek index
SELECT
    TABLE_NAME,
    INDEX_NAME,
    GROUP_CONCAT(COLUMN_NAME ORDER BY SEQ_IN_INDEX) AS COLUMNS,
    NON_UNIQUE
FROM
    information_schema.STATISTICS
WHERE
    TABLE_SCHEMA = 'disjaya_dashboard' AND INDEX_NAME != 'PRIMARY'
GROUP BY
    TABLE_NAME, INDEX_NAME, NON_UNIQUE
ORDER BY
    TABLE_NAME, INDEX_NAME;

======================= compare data dan table mariadb dan postgresql ===========================================

--- mariadb

SET group_concat_max_len = 1000000;

SELECT GROUP_CONCAT(
    CONCAT('SELECT ''', TABLE_NAME, ''' AS table_name, COUNT(*) AS row_count FROM `', TABLE_SCHEMA, '`.`', TABLE_NAME, '`')
    SEPARATOR ' UNION ALL '
) INTO @sql
FROM INFORMATION_SCHEMA.TABLES
WHERE TABLE_SCHEMA = 'disjaya_dashboard'
  AND TABLE_TYPE = 'BASE TABLE';

PREPARE stmt FROM @sql;
EXECUTE stmt;
DEALLOCATE PREPARE stmt;

--- postgresql

SELECT n.nspname AS schema_name,
       c.relname AS table_name,
       c.reltuples AS estimated_row_count
FROM pg_class c
JOIN pg_namespace n ON n.oid = c.relnamespace
WHERE c.relkind = 'r'
  AND n.nspname NOT IN ('pg_catalog', 'information_schema')
ORDER BY schema_name, table_name;
