@echo off
setlocal

REM Definisikan variabel
set SRC_DIR=E:\BackupDB\ProRBA2PLN
set DEST_DIR=/Backup_Database/Database/production/smarter
set SERVER=icon@10.14.152.62
set PORT=6969
set PASSWORD=R4in!Sc0mFOrT1nG

REM Temukan file .bak terbaru
set "newest_file="
for %%f in ("%SRC_DIR%\*.bak") do (
    if not defined newest_file (
        set "newest_file=%%f"
    ) else (
        if "%%~tf" GTR "%newest_file%" (
            set "newest_file=%%f"
        )
    )
)

REM Menyalin file .bak terbaru menggunakan pscp
if defined newest_file (
    echo Menyalin %newest_file% ke %SERVER%:%DEST_DIR% menggunakan port %PORT%
    pscp -P %PORT% -pw %PASSWORD% "%newest_file%" %SERVER%:%DEST_DIR%
) else (
    echo Tidak ada file .bak ditemukan di %SRC_DIR%.
)

endlocal
echo Selesai menyalin file.
timeout /t 5 /nobreak >nul
exit
