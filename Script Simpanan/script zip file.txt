@echo off
setlocal

REM Definisikan variabel
set SRC_DIR=E:\DATABASE\BACKUP\BACKUP_NEW
set DEST_DIR=E:\DATABASE\BACKUP\BACKUP_NEW

REM Mendapatkan tanggal dalam format YYYYMMDD
for /f "tokens=1-3 delims=/- " %%a in ('wmic os get localdatetime ^| find "."') do (
    set "date_part=%%a"
)

REM Mendapatkan jam saat ini dalam format HH (24 jam)
set "hour_part=%date_part:~8,2%"

REM Mendapatkan tanggal dalam format YYYYMMDD
set "date_part=%date_part:~0,4%%date_part:~4,2%%date_part:~6,2%"

REM Format tanggal dan waktu untuk nama file ZIP
set "timestamp=%date_part%_%hour_part%"

REM Nama file ZIP dengan timestamp
set ZIP_FILE=%DEST_DIR%\backupalldbsilm_%timestamp%.zip

REM Path ke WinRAR (pastikan path ini sesuai dengan lokasi instalasi WinRAR di sistem Anda)
set WINRAR_PATH="C:\Program Files\WinRAR\WinRAR.exe"

REM Membuat ZIP dari seluruh folder dan file .bak di SRC_DIR
echo Membuat ZIP dari seluruh folder dan file .bak di %SRC_DIR%
%WINRAR_PATH% a -r -afzip "%ZIP_FILE%" "%SRC_DIR%\*"

REM Periksa apakah file ZIP berhasil dibuat
if exist "%ZIP_FILE%" (
    echo ZIP berhasil dibuat: %ZIP_FILE%

    REM Menghapus semua file .bak di SRC_DIR
    echo Menghapus file .bak di %SRC_DIR%
    for /r "%SRC_DIR%" %%f in (*.bak) do del "%%f"
) else (
    echo Gagal membuat file ZIP.
)

endlocal
echo Proses selesai. Script akan keluar dalam 5 detik.
timeout /t 5 /nobreak >nul
exit
