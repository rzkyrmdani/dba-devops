@echo off
setlocal

:: ===============================
:: KONFIGURASI DATABASE & BACKUP
:: ===============================
set "SQL_USER=backup_db"
set "SQL_PASS=Plnpusat@2025!"
set "SQL_SERVER=WIN-4OM4B4TF6R2\MAXICOPLNDB"
set "BACKUP_DIR=E:\Backup_DB\Full"
set "SQLCMD_PATH=C:\Program Files\Microsoft SQL Server\Client SDK\ODBC\170\Tools\Binn\SQLCMD.EXE"
set "PSCP_PATH=C:\Program Files\PuTTY\pscp.exe"
set "WINRAR_PATH=C:\Program Files\WinRAR\WinRAR.exe"
set "MC_PATH=C:\Program Files\MC\mc.exe"
set "ZIP_PASSWORD=Iconplnkorporat1"
set "BACKUP_METHOD=SQLSERVER"
set "NOTIF_SCRIPT=E:\Script\reportlogdb.ps1"

:: ===============================
:: SERVER SCP
:: ===============================
set "SCP1_PORT=6969"
set "SCP1_USER=icon"
set "SCP1_PASS=R4in!Sc0mFOrT1nG"
set "SCP1_IP=10.14.152.62"
set "SCP1_TARGET=/Backup_Database/Database/production/maxico_pln"

set "SCP2_PORT=22"
set "SCP2_USER=sqlserver"
set "SCP2_PASS=Icon@thn2025!"
set "SCP2_IP=10.14.113.34"
set "SCP2_TARGET=/backupfull_sqlserver/maxico_pln/backup_full"

:: ===============================
:: MINIO
:: ===============================
set "MINIO_ALIAS=myminio/backup-dump/prod/maxico-pln/dump/"

:: ===============================
:: PREPARE
:: ===============================
if not exist "%BACKUP_DIR%" mkdir "%BACKUP_DIR%"

for /f %%i in ('powershell -NoLogo -Command "Get-Date -Format yyyyMMdd_HHmmss"') do set DATETIME=%%i
set "ZIP_NAME=Prod_All_DBMaxicoPLN_BackupFull_%DATETIME%.zip"
set "ZIP_PATH=%BACKUP_DIR%\%ZIP_NAME%"

:: ===============================
:: START TIME
:: ===============================
for /f %%i in ('powershell -NoLogo -Command "Get-Date -Format yyyy-MM-dd_HH:mm:ss"') do set START_TIME=%%i

echo [INFO] Memulai proses FULL BACKUP semua database user...

:: ===============================
:: BACKUP SEMUA DATABASE USER
:: ===============================
"%SQLCMD_PATH%" -S %SQL_SERVER% -U %SQL_USER% -P %SQL_PASS% -Q ^
"DECLARE @name NVARCHAR(256), @fileName NVARCHAR(512), @fileDate NVARCHAR(20), @path NVARCHAR(512); ^
SET @path = N'%BACKUP_DIR%\'; ^
SET @fileDate = CONVERT(VARCHAR(20), GETDATE(), 112) + '_' + REPLACE(CONVERT(VARCHAR(8), GETDATE(), 108), ':', ''); ^
DECLARE db_cursor CURSOR FOR SELECT name FROM master.sys.databases WHERE name NOT IN ('master','model','msdb','tempdb'); ^
OPEN db_cursor FETCH NEXT FROM db_cursor INTO @name ^
WHILE @@FETCH_STATUS = 0 ^
BEGIN SET @fileName = @path + @name + '_FULL_' + @fileDate + '.bak'; ^
EXEC('BACKUP DATABASE [' + @name + '] TO DISK = ''' + @fileName + ''' WITH INIT, COMPRESSION'); ^
FETCH NEXT FROM db_cursor INTO @name END ^
CLOSE db_cursor DEALLOCATE db_cursor"

:: ===============================
:: VALIDASI FILE BAK
:: ===============================
dir "%BACKUP_DIR%\*.bak" >nul 2>&1
if errorlevel 1 goto FAILED

:: ===============================
:: ZIP FILE
:: ===============================
"%WINRAR_PATH%" a -ep1 -ibck -m5 -r -p%ZIP_PASSWORD% "%ZIP_PATH%" "%BACKUP_DIR%\*.bak"

:: ===============================
:: HITUNG SIZE ZIP
:: ===============================
for %%A in ("%ZIP_PATH%") do set ZIP_SIZE_BYTES=%%~zA
for /f %%i in ('powershell -NoLogo -Command "[math]::Round(%ZIP_SIZE_BYTES%/1MB,2)"') do set ZIP_SIZE_MB=%%i

:: ===============================
:: CLEANUP BAK
:: ===============================
del /q "%BACKUP_DIR%\*.bak"

:: ===============================
:: UPLOAD PARALEL (SERVER 1 & 2)
:: ===============================
echo [INFO] Upload paralel ke Server 1 & Server 2...

start "SCP1" cmd /c ^
""%PSCP_PATH%" -P %SCP1_PORT% -pw "%SCP1_PASS%" "%ZIP_PATH%" %SCP1_USER%@%SCP1_IP%:%SCP1_TARGET%"

start "SCP2" cmd /c ^
""%PSCP_PATH%" -P %SCP2_PORT% -pw "%SCP2_PASS%" "%ZIP_PATH%" %SCP2_USER%@%SCP2_IP%:%SCP2_TARGET%"

:: Tunggu SCP selesai
timeout /t 20 /nobreak >nul

:: ===============================
:: UPLOAD KE MINIO
:: ===============================
echo [INFO] Upload ke MinIO...
"%MC_PATH%" --insecure cp "%ZIP_PATH%" %MINIO_ALIAS%
set MINIO_STATUS=%ERRORLEVEL%

:: ===============================
:: STATUS FINAL
:: ===============================
if %MINIO_STATUS% EQU 0 (
    set "BACKUP_STATUS=SUCCESS"
    set "BACKUP_DESC=Backup Full Database MAXICO_PLN successfully completed."
) else (
    set "BACKUP_STATUS=FAILED"
    set "BACKUP_DESC=Backup Full Database MAXICO_PLN FAILED (MinIO error)."
)

goto REPORT

:FAILED
set "BACKUP_STATUS=FAILED"
set "BACKUP_DESC=Backup Full Database MAXICO_PLN FAILED."

:REPORT
:: ===============================
:: END TIME
:: ===============================
for /f %%i in ('powershell -NoLogo -Command "Get-Date -Format yyyy-MM-dd_HH:mm:ss"') do set END_TIME=%%i

:: ===============================
:: NOTIF REPORT
:: ===============================
powershell -NoProfile -ExecutionPolicy Bypass -File "%NOTIF_SCRIPT%" ^
 -Status "%BACKUP_STATUS%" ^
 -StartTime "%START_TIME%" ^
 -EndTime "%END_TIME%" ^
 -Description "%BACKUP_DESC%" ^
 -Size "%ZIP_SIZE_MB%MB" ^
 -Method "%BACKUP_METHOD%"

:: ===============================
:: FINISH
:: ===============================
if "%BACKUP_STATUS%"=="SUCCESS" (
    echo [INFO] Backup MAXICO_PLN sukses.
    exit /b 0
) else (
    echo [ERROR] Backup MAXICO_PLN gagal.
    exit /b 1
)
